当前 src/App.jsx（完整代码，可直接替换）

import { useState, useEffect } from 'react'
import Papa from 'papaparse'
import './App.css'function App() {
  const [isFullscreen, setIsFullscreen] = useState(false)
  const [isActive, setIsActive] = useState(false)
  const [participants, setParticipants] = useState([])
  const [currentIndex, setCurrentIndex] = useState(0)
  const [settings, setSettings] = useState({})
  const [loading, setLoading] = useState(true)  // 读取配置和名单
  useEffect(() => {
    fetch('/configuration/settings.json')
      .then(res => res.json())
      .then(data => {
        // 自动过滤注释键
        const cleanData = Object.fromEntries(
          Object.entries(data).filter(([key]) => !key.startsWith('//'))
        )
        setSettings(cleanData)    fetch(cleanData.participants)
      .then(csvRes => csvRes.text())
      .then(csvText => {
        Papa.parse(csvText, {
          header: true,
          skipEmptyLines: true,
          complete: (result) => {
            const filtered = result.data.filter(row => row.id && !row.id.trim().startsWith('#'))
            // 限制列数
            const limited = filtered.map(row => {
              const limitedRow = {}
              Object.keys(row).slice(0, cleanData.read_fields_max).forEach(key => {
                limitedRow[key] = row[key]
              })
              return limitedRow
            })
            setParticipants(limited)
            setLoading(false)
          }
        })
      })
  })
  .catch(err => {
    console.error('读取失败', err)
    setLoading(false)
  })  }, [])  // 滚动逻辑
  useEffect(() => {
    if (!isFullscreen || participants.length === 0) returnlet interval
if (isActive) {
  interval = setInterval(() => {
    setCurrentIndex(prev => (prev + 1) % participants.length)
  }, 1500)
}
return () => clearInterval(interval)  }, [isFullscreen, isActive, participants])  // 键盘 + 全屏事件
  useEffect(() => {
    const handleKeyDown = (e) => {
      if (!isFullscreen) return
      if (e.key === ' ') {
        e.preventDefault()
        setIsActive(prev => !prev)
      }
      if (e.key === 'Escape') {
        e.preventDefault()
        if (document.exitFullscreen) document.exitFullscreen()
      }
    }const handleFullscreenChange = () => {
  const isFull = !!document.fullscreenElement
  setIsFullscreen(isFull)
  if (!isFull) setIsActive(false)
}

window.addEventListener('keydown', handleKeyDown)
document.addEventListener('fullscreenchange', handleFullscreenChange)

return () => {
  window.removeEventListener('keydown', handleKeyDown)
  document.removeEventListener('fullscreenchange', handleFullscreenChange)
}  }, [isFullscreen])  const enterFullscreen = () => {
    document.documentElement.requestFullscreen?.()
  }  if (loading) return <div className="main-screen">加载配置中...</div>  if (!isFullscreen) {
    return (
      <div className="main-screen">
        <button className="start-button" onClick={enterFullscreen}>
          开始抽奖
        </button>
        <p className="hint">名单人数：{participants.length} 人</p>
      </div>
    )
  }  const current = participants[currentIndex] || {}
  const displayText = Array(settings.display_fields || 2)
    .fill('')
    .map((_, i) => Object.values(current)[i] || '')
    .filter(Boolean)
    .join(' ')  return (
    <div className="fullscreen">
      <h1 className={isActive ? 'blink' : ''}>
        下一个幸运儿：{displayText || '加载中...'}
      </h1>  <div className="tips">
    空格键：{isActive ? '暂停' : '开始'}　　Esc：退出全屏
  </div>
</div>  )
}export default App


核心逻辑总结（按你描述）初始化 & 配置读取读取 ./configuration/settings.json
读取其中 participants 字段指向的 CSV 文件
CSV 每行最多只取前 read_fields_max 个字段
过滤掉无效行（id 为空 或 以 # 开头）

奖项顺序（固定，从低到高）

hotpot-prize → 6th-prize → 5th-prize → 4th-prize → 3rd-prize → 2nd-prize → 1st-prize → (可选) grand-prize

奖项跳过 & 连续性规则校验（非常关键的一段）当程序启动/读取配置后，要做以下检查（任一不通过 → 弹窗报错 + “开始抽奖”按钮禁用）：从 hotpot-prize 开始往后扫描，第一个非0的奖项才是“真正要开始抽的第一轮”
从这个第一个非0奖项一直到 1st-prize（甚至到 grand-prize），中间不允许再出现0
在这个“有效奖项区间”内，获奖人数必须非递增（后面奖项人数 ≤ 前面奖项人数）举例：允许 7 → 5 → 3 → 2 → 1
不允许 5 → 7（人数变多）
不允许 4 → 0 → 2（中间断档）

典型合法配置示例：

"hotpot-prize": 8,
"6th-prize":    0,     // 跳过
"5th-prize":    0,     // 跳过
"4th-prize":    6,
"3rd-prize":    4,
"2nd-prize":    2,
"1st-prize":    1,
"grand-prize":  1      // 或 0

→ 实际抽奖顺序：四等奖(6人) → 三等奖(4) → 二等奖(2) → 一等奖(1) → 特等奖(1)
界面 & 交互流程（每轮）显示：“下面抽取 ××奖   剩余 ×× 名额”
按空格 → 开始快速滚动名单（只显示前 display_fields 个字段）
再次按空格 → 停止滚动 → 显示中奖者（放大、高亮、恭喜动画等）
确认中奖后（可加个“确认”按键或自动几秒后）：记录该人已中奖（永久排除）
当前奖项剩余名额 -1
如果该奖项名额归零 → 自动（或手动）进入下一奖项

全部奖项抽完 → 显示大字“抽奖结束” + 可选的全中奖名单回顾




当前确认的核心规则（2026版抽奖流程）序号
规则描述
细节说明
1
奖项顺序 & 跳过0
hotpot → 6th → 5th → 4th → 3rd → 2nd → 1st → (可选 grand)
数量=0的奖项直接跳过，不抽那一轮
2
每轮抽奖方式
整轮一次性抽完该奖项所有名额（不是一个一个抽）
例如四等奖有6人 → 这一轮滚动 → 暂停 → 一次性显示6个中奖人
3
按空格行为（循环）
状态A（显示“下面抽取XX奖 名额XXX”） → 空格 → 开始滚动
滚动中 → 空格 → 停止 + 显示本轮中奖名单（多个人）
显示中奖名单 → 空格 → 清空名单 → 回到状态A（显示下一轮提示或本轮如果没抽完）
4
已中奖排除
一旦某人出现在任何一轮的中奖名单里，此人永久从后续所有滚动名单中排除
5
名额显示
